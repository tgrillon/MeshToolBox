37void Camera::UpdateView()

46std::pair<float, float> Camera::PanSpeed() const
47{
48 float x = std::min(m_ViewportWidth / 1000.0f, 2.4f); // max = 2.4f
49 float xFactor = 0.0366f * (x * x) - 0.1778f * x + 0.3021f;
50
51 float y = std::min(m_ViewportHeight / 1000.0f, 2.4f); // max = 2.4f
52 float yFactor = 0.0366f * (y * y) - 0.1778f * y + 0.3021f;
53
54 return { xFactor * m_TranslationSpeed, yFactor * m_TranslationSpeed };
55}

57float Camera::ZoomSpeed() const
58{
59 float distance = m_Distance * 0.2f;

66void Camera::OnUpdate(const float ts)
67{
68 if(Input::IsKeyPressed(Key::LeftAlt))
69 {
70 const Vec2& mouse{ Input::GetMouseX(), Input::GetMouseY() };
71 Vec2 delta = (mouse - m_InitialMousePosition) * 0.003f;
72 m_InitialMousePosition = mouse;
73
74 if(Input::IsMouseButtonPressed(Mouse::ButtonMiddle))
75 MousePan(delta);
76 else if(Input::IsMouseButtonPressed(Mouse::ButtonLeft))
77 MouseRotate(delta);
78 else if(Input::IsMouseButtonPressed(Mouse::ButtonRight))
79 {
80 if(std::abs(delta.x) > std::abs(delta.y * m_AspectRatio))
81 MouseZoom(delta.x);
82 else
83 MouseZoom(delta.y * m_AspectRatio);
84 }

103void Camera::MousePan(const Vec2& delta)
104{
105 auto [xSpeed, ySpeed] = PanSpeed();
106 m_Translation.x += -delta.x * xSpeed * m_Distance;
107 m_Translation.y += delta.y * ySpeed * m_Distance;

110void Camera::MouseRotate(const Vec2& delta)
111{
112 float yawSign = GetUpDirection().y < 0 ? -1.0f : 1.0f;
113 m_Yaw += yawSign * delta.x * m_RotationSpeed;
